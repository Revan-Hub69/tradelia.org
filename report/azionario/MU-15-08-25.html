// --- TABS/Single-View --- const tabsBar = document.getElementById('section-tabs'); const scroller = document.getElementById('tabs-scroller'); const tabs = Array.from(document.querySelectorAll('.sv-tab')); const panels = Array.from(document.querySelectorAll('.sv-panel')); const tabMap = new Map(tabs.map(b => [b.dataset.target, b])); const panelMap = new Map(panels.map(p => [p.id, p])); function prefersReducedMotion(){ return matchMedia('(prefers-reduced-motion: reduce)').matches; } function updateOffsets(){ const topOffset = header.getBoundingClientRect().height; tabsBar.style.top = topOffset + 'px'; const extra = tabsBar.getBoundingClientRect().height + 8; panels.forEach(p => p.style.scrollMarginTop = (topOffset + extra) + 'px'); } function setActive(id, pushHash=true, doScroll=true){ if (!panelMap.has(id)) return; // Tabs aria + tabindex tabs.forEach(t => { const sel = (t.dataset.target === id); t.classList.toggle('active', sel); t.setAttribute('aria-selected', sel ? 'true' : 'false'); t.tabIndex = sel ? 0 : -1; }); // Panels visibility panels.forEach(p => { const vis = (p.id === id); p.classList.toggle('hidden', !vis); p.toggleAttribute('hidden', !vis); }); // URL + storage if (pushHash) { try { history.replaceState(null, '', '#'+id); } catch(e){} } try { localStorage.setItem('tradelia_active_section', id); } catch(e){} // Scroll pagina if (doScroll){ const topOffset = header.getBoundingClientRect().height; const tabsH = tabsBar.getBoundingClientRect().height + 8; const y = panelMap.get(id).getBoundingClientRect().top + scrollY - (topOffset + tabsH); scrollTo({ top: y, behavior: prefersReducedMotion() ? 'auto' : 'smooth' }); } // Assicurati che il tab attivo sia visibile nello scroller const activeTab = tabMap.get(id); activeTab?.scrollIntoView({ inline: 'nearest', block: 'nearest', behavior: prefersReducedMotion() ? 'auto' : 'smooth' }); // Live region + TOC highlight document.getElementById('live-region').textContent = `Sezione: ${id}`; tocLinks.forEach(l => l.classList.remove('bg-sky-50','text-sky-700','font-semibold')); document.querySelector(`#toc-drawer .toc-link[href="#${id}"]`)?.classList.add('bg-sky-50','text-sky-700','font-semibold'); if (window.lucide && lucide.createIcons) lucide.createIcons(); } window.setActive = setActive; // Click tabs tabs.forEach(t => t.addEventListener('click', () => setActive(t.dataset.target))); // Frecce tastiera (a11y): ← → tabsBar.addEventListener('keydown', (e) => { if (e.key !== 'ArrowRight' && e.key !== 'ArrowLeft') return; const current = document.activeElement.closest('.sv-tab'); if (!current) return; e.preventDefault(); const idx = tabs.indexOf(current); const nextIdx = e.key === 'ArrowRight' ? Math.min(idx + 1, tabs.length - 1) : Math.max(idx - 1, 0); const next = tabs[nextIdx]; next.focus(); setActive(next.dataset.target, true, false); }); function initActiveFromHash(){ const saved = localStorage.getItem('tradelia_active_section'); const hash = (location.hash || '').replace('#',''); const target = panelMap.has(hash) ? hash : (panelMap.has(saved) ? saved : 'sim-op'); setActive(target, false, false); } updateOffsets(); addEventListener('resize', updateOffsets); initActiveFromHash();
